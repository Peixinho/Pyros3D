# the device layer test scene - a standalone build (not part of the engine's yet):
#   cmake -S tools/device_test -B build_d2 -DDAWN_DIR=<a Dawn release>  &&  cmake --build build_d2
cmake_minimum_required(VERSION 3.20)
if(APPLE)
  project(device_test C CXX OBJC OBJCXX)
else()
  project(device_test C CXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(Dawn REQUIRED PATHS ${DAWN_DIR}/lib/cmake/Dawn NO_DEFAULT_PATH)
find_package(SDL2 REQUIRED)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
# the test's shader as SPIR-V, built here (glslc), for the Vulkan backend
find_program(GLSLC glslc REQUIRED)
foreach(stage vert frag)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/test.${stage}.inc
    COMMAND ${GLSLC} -O -mfmt=num ${CMAKE_CURRENT_SOURCE_DIR}/test.${stage} -o ${CMAKE_CURRENT_BINARY_DIR}/test.${stage}.inc
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/test.${stage})
endforeach()
set(FETCH ${ROOT}/.fetchcontent-cache)
# (volk and VMA as the engine fetches them: the engine's .fetchcontent-cache, or VOLK_DIR / VMA_DIR)
if(NOT VOLK_DIR)
  set(VOLK_DIR ${FETCH}/volk-src)
endif()
if(NOT VMA_DIR)
  set(VMA_DIR ${FETCH}/vulkanmemoryallocator-src)
endif()
set(PYROS_METAL_SRC)
if(APPLE)
  set(PYROS_METAL_SRC ${ROOT}/src/Pyros3D/Rendering/Device/MetalDevice.mm)
endif()
add_executable(device_test main.cpp ${ROOT}/src/Pyros3D/Rendering/Device/WebGPUDevice.cpp ${ROOT}/src/Pyros3D/Rendering/Device/VulkanDevice.cpp
  ${PYROS_METAL_SRC} ${VOLK_DIR}/volk.c ${CMAKE_CURRENT_BINARY_DIR}/test.vert.inc ${CMAKE_CURRENT_BINARY_DIR}/test.frag.inc)
find_package(Vulkan QUIET)
target_include_directories(device_test PRIVATE ${ROOT}/include ${SDL2_INCLUDE_DIRS} ${CMAKE_CURRENT_BINARY_DIR}
  ${VOLK_DIR} ${VMA_DIR}/include ${Vulkan_INCLUDE_DIRS} /opt/homebrew/include)
if(APPLE)
  target_compile_definitions(device_test PRIVATE VK_USE_PLATFORM_METAL_EXT VK_NO_PROTOTYPES)
elseif(WIN32)
  target_compile_definitions(device_test PRIVATE VK_USE_PLATFORM_WIN32_KHR VK_NO_PROTOTYPES NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_compile_definitions(device_test PRIVATE VK_USE_PLATFORM_XLIB_KHR VK_NO_PROTOTYPES)
endif()
if(TARGET SDL2::SDL2)
  target_link_libraries(device_test PRIVATE dawn::webgpu_dawn SDL2::SDL2 $<TARGET_NAME_IF_EXISTS:SDL2::SDL2main>)
else()
  target_link_libraries(device_test PRIVATE dawn::webgpu_dawn ${SDL2_LIBRARIES})
endif()
if(APPLE)
  target_link_libraries(device_test PRIVATE "-framework Metal" "-framework QuartzCore" "-framework IOSurface" "-framework Foundation" "-framework Cocoa" "-framework IOKit")
  # (the Metal backend turns SPIR-V into the Metal shading language with SPIRV-Cross - Homebrew's, as the engine's)
  execute_process(COMMAND brew --prefix spirv-cross OUTPUT_VARIABLE SPIRV_CROSS_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE)
  target_include_directories(device_test PRIVATE ${SPIRV_CROSS_PREFIX}/include)
  target_link_libraries(device_test PRIVATE ${SPIRV_CROSS_PREFIX}/lib/libspirv-cross-msl.a ${SPIRV_CROSS_PREFIX}/lib/libspirv-cross-glsl.a ${SPIRV_CROSS_PREFIX}/lib/libspirv-cross-core.a)
  set_source_files_properties(${PYROS_METAL_SRC} PROPERTIES COMPILE_FLAGS "-fobjc-arc")
endif()
